import limen.graphics.GraphicsDriver;
import limen.graphics.renderer.opengl.OpenGL;
import limen.graphics.renderer.opengl.command.Commands;
import limen.platform.Platform;
import limen.platform.event.Event;
import limen.platform.event.WindowStateChange;
import limen.platform.window.Window;
import limen.platform.window.WindowFlags;

class Main {
    static function main() {
        // require OpenGL and disable driver fallback
        Platform.init(GraphicsDriver.OpenGL, [GraphicsDriver.OpenGL]);

        // change the title and starting size here
        final window = Window.create({
            title: "LIMEN GL",
            width: 960,
            height: 540,
            flags: WindowFlags.SDL_WINDOW_OPENGL,
            resizable: true
        });
        // make this context current before issuing OpenGL commands
        final context = OpenGL.create(window, {minimumMajor: 3, minimumMinor: 3});
        context.makeCurrent();

        // keep the window responsive until it closes
        final event = new Event();
        var running = true;

        while (running) {
            while (Platform.pollEvent(event)) {
                switch (event.type) {
                    case Quit:
                        running = false;
                    case WindowState if (event.windowId == window.id && event.state == WindowStateChange.Close):
                        running = false;
                    default:
                }
            }

            if (!running) break;

            // change the first three values (red, green, blue) from 0 to 1
            Commands.clearColor(0.12, 0.25, 0.48, 1);
            Commands.clear(Commands.COLOR_BUFFER_BIT);
            // swap buffers to show the cleared frame
            context.present();
            Platform.delay(8);
        }

        // release the context before destroying its window
        context.destroy();
        window.destroy();
        Platform.quit();
    }
}
