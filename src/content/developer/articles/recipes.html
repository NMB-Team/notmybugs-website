import limen.platform.Platform;
import limen.platform.display.Display;
import limen.platform.system.Paths;
import limen.platform.system.System;
import sys.io.File;

class Main {
    static function main() {
        Platform.init(None);

        for (screen in Display.all()) {
            Sys.println('${screen.name}: ${screen.width}x${screen.height}');
        }

        Sys.println('System theme: ${System.theme}');
        final directory = Paths.preference("Limen Examples", "Target Rush");

        if (directory != null) {
            File.saveContent(directory + "record.txt", "10\n");
            Sys.println(directory + "record.txt");
        }

        Platform.quit();
    }
}
