import limen.graphics.GraphicsDriver;
import limen.graphics.renderer.d3d12.D3D12;
import limen.graphics.renderer.d3d12.DX12Core.ClearColor;
import limen.graphics.renderer.d3d12.command.Commands;
import limen.graphics.renderer.d3d12.descriptor.DescriptorHeap;
import limen.graphics.renderer.d3d12.internal.D3D12Bindings;
import limen.graphics.renderer.d3d12.resource.Resources;
import limen.platform.Platform;
import limen.platform.event.Event;
import limen.platform.event.WindowStateChange;
import limen.platform.window.Window;

class Main {
    static function main():Void {
        // require Direct3D 12 and disable driver fallback
        Platform.init(GraphicsDriver.D3D12, [GraphicsDriver.D3D12]);

        // change the title and starting size here
        final window = Window.create({
            title: "LIMEN Direct3D 12",
            width: 800,
            height: 450,
            resizable: false
        });

        final driver = D3D12.create(window, new haxe.EnumFlags());
        final queue = new CommandQueue(DIRECT);
        // create a swapchain with two back buffers
        driver.resize(queue, window.width, window.height, 2, R8G8B8A8_UNORM);

        final buffer = D3D12Bindings.getBackBuffer(D3D12Bindings.getCurrentBackBufferIndex());
        // reserve one render target descriptor
        final heapDesc = new DescriptorHeapDesc();
        heapDesc.type = RTV;
        heapDesc.numDescriptors = 1;
        final heap = new DescriptorHeap(heapDesc);
        final target = heap.getHandle(false);
        D3D12Bindings.createRenderTargetView(buffer, null, target);

        final allocator = new CommandAllocator(DIRECT);
        final commands = new CommandList(DIRECT, allocator, null);

        // transition the buffer so the GPU can write to it
        final barrier = new ResourceBarrier();
        barrier.resource = buffer;
        barrier.subResource = -1;
        barrier.stateBefore = PRESENT;
        barrier.stateAfter = RENDER_TARGET;
        commands.resourceBarrier(barrier);

        final color = new ClearColor();
        // change red, green, and blue from 0 to 1
        color.r = 0.12;
        color.g = 0.25;
        color.b = 0.48;
        color.a = 1;
        commands.clearRenderTargetView(target, color);

        // transition the back buffer for presentation
        barrier.stateBefore = RENDER_TARGET;
        barrier.stateAfter = PRESENT;
        commands.resourceBarrier(barrier);
        commands.close();

        // signal this fence after the queued frame
        final frameDone = 1;
        final fence = new Fence(0, NONE);
        queue.executeCommandList(commands);
        queue.present(1);
        queue.signal(fence, frameDone);

        // keep the window responsive until it closes
        final event = new Event();
        var running = true;

        while (running) {
            while (Platform.pollEvent(event)) {
                switch (event.type) {
                    case Quit:
                        running = false;
                    case WindowState if (event.windowId == window.id && event.state == WindowStateChange.Close):
                        running = false;
                    default:
                }
            }

            if (running)
                Platform.delay(10);
        }

        // wait for the GPU before releasing resources
        while (haxe.Int64.compare(fence.getValue(), frameDone) < 0)
            Platform.delay(1);

        buffer.release();
        (heap : Dx12Resource).release();
        (cast commands : Dx12Resource).release();
        (cast allocator : Dx12Resource).release();
        (cast fence : Dx12Resource).release();
        (cast queue : Dx12Resource).release();
        driver.destroy();
        window.destroy();
        Platform.quit();
    }
}
