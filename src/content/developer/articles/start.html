import limen.graphics.GraphicsDriver;
import limen.platform.Platform;
import limen.platform.window.Window;
import limen.platform.event.Event;
import limen.platform.event.EventType;
import limen.platform.event.WindowStateChange;

class Main {
    static function main() {
        Platform.init(GraphicsDriver.None);

        final window = Window.create({
            title: "Hello, LIMEN!",
            width: 800,
            height: 500,
            resizable: true
        });

        final event = new Event();
        var running = true;

        while (running) {
            while (Platform.pollEvent(event)) {
                if (event.type == EventType.Quit ||
                    (event.type == EventType.WindowState &&
                     event.windowId == window.id && event.state == WindowStateChange.Close)) {
                    running = false;
                }
            }
            Platform.delay(8);
        }

        window.destroy();
        Platform.quit();
    }
}
