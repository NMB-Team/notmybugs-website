Main.hx
HAXE 
final event = new Event();
var running = true;

while (running) {
    while (Platform.pollEvent(event)) {
        switch (event.type) {
            case EventType.Quit:
                running = false;
            case EventType.MouseMove:
                trace(event.mouseX, event.mouseY);
            case EventType.KeyDown:
                trace(event.keyCode);
            default:
        }
    }
    Platform.delay(8);
}
