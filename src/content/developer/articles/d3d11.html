import limen.graphics.GraphicsDriver;
import limen.graphics.renderer.d3d11.D3D11;
import limen.graphics.renderer.d3d11.DX11Core.Format;
import limen.graphics.renderer.d3d11.DX11Core.PresentFlags;
import limen.graphics.renderer.d3d11.command.Commands;
import limen.graphics.renderer.d3d11.descriptor.ResourceViews;
import limen.graphics.renderer.d3d11.internal.D3D11Bindings;
import limen.platform.Platform;
import limen.platform.event.Event;
import limen.platform.event.WindowStateChange;
import limen.platform.window.Window;

class Main {
    static function main():Void {
        // require Direct3D 11 and disable driver fallback
        Platform.init(GraphicsDriver.D3D11, [GraphicsDriver.D3D11]);

        // change the title and starting size here
        final window = Window.create({
            title: "LIMEN Direct3D 11",
            width: 800,
            height: 450,
            resizable: false
        });

        final driver = D3D11.create(window, Format.R8G8B8A8_UNORM);
        // get the back buffer and create a view the GPU can clear
        final buffer = D3D11Bindings.getBackBuffer();
        final target = ResourceViews.createRenderTargetView(buffer);
        // the view keeps its own reference to the back buffer
        buffer.release();

        // change the red, green, and blue values from 0 to 1
        Commands.clearColor(target, 0.12, 0.25, 0.48, 1);
        // show the frame at the next screen refresh
        driver.present(1, PresentFlags.None);

        // keep the window responsive until it closes
        final event = new Event();
        var running = true;

        while (running) {
            while (Platform.pollEvent(event)) {
                switch (event.type) {
                    case Quit:
                        running = false;
                    case WindowState if (event.windowId == window.id && event.state == WindowStateChange.Close):
                        running = false;
                    default:
                }
            }

            if (running)
                Platform.delay(10);
        }

        // release the render target view before destroying the driver
        target.release();
        driver.destroy();
        window.destroy();
        Platform.quit();
    }
}
