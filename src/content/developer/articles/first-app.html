import limen.graphics.GraphicsDriver;
import limen.platform.Platform;
import limen.platform.event.Event;
import limen.platform.event.EventType;
import limen.platform.event.WindowStateChange;
import limen.platform.system.Clipboard;
import limen.platform.system.dialog.FileDialog;
import limen.platform.system.dialog.FileDialogResult;
import limen.platform.window.Window;
import sys.io.File;
import sys.thread.Deque;

class Main {
    static function main() {
        Platform.init(GraphicsDriver.None);

        final window = Window.create({
            title: "Text counter | O: open, C: copy",
            width: 700,
            height: 240
        });
        final results = new Deque<FileDialogResult>();
        final event = new Event();

        var running = true;
        var choosing = false;
        var report:Null<String> = null;

        while (running) {
            while (running && Platform.pollEvent(event)) {
                switch (event.type) {
                    case Quit:
                        running = false;
                    case WindowState if (event.windowId == window.id && event.state == WindowStateChange.Close):
                        running = false;
                    case KeyDown if (event.windowId == window.id && !event.keyRepeat && !choosing):
                        switch (event.scanCode) {
                            case 18: // O
                                choosing = true;
                                FileDialog.openFile(result -> results.add(result), {
                                    parent: window,
                                    filters: [{name: "Text files", pattern: "txt;md"}]
                                });
                            case 6 if (report != null): // C
                                window.title = Clipboard.setText(report)
                                    ? report + " | Copied"
                                    : report + " | Copy failed";
                            default:
                        }
                    default:
                }
            }

            if (!running) break;

            final result = results.pop(false);
            if (result != null) {
                choosing = false;

                switch (result) {
                    case Selected(paths, _):
                        try {
                            final content = File.getContent(paths[0]);
                            final words = [for (word in ~/\s+/g.split(content)) if (word != "") word];
                            report = '${words.length} words';
                            window.title = report + " | O: open, C: copy";
                        } catch (error:Dynamic) {
                            report = null;
                            window.title = "Could not read the file | O: try again";
                            Sys.println(Std.string(error));
                        }
                    case Cancelled:
                    case Failed(message):
                        window.title = "File dialog failed | O: try again";
                        Sys.println(message);
                }
            }

            Platform.delay(8);
        }

        window.destroy();
        Platform.quit();
    }
}
